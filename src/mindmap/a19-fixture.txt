(MarkdownRenderer as any).render = async (_app: any, text: string, host: HTMLElement) => {
  host.replaceChildren();
  const pattern = /\[([^\]]+)\]\(([^)]+)\)/g;
  let match: RegExpExecArray | null, start = 0;
  while ((match = pattern.exec(text))) {
    host.append(document.createTextNode(text.slice(start, match.index)));
    const link = document.createElement('a'); link.textContent = match[1]; link.href = match[2]; host.append(link); start = pattern.lastIndex;
  }
  host.append(document.createTextNode(text.slice(start)));
};

function draw(host: HTMLElement, from: string, to: string) {
  const source = graphNode(host, from).getBoundingClientRect(), dest = graphNode(host, to).getBoundingClientRect();
  const x1 = source.right - 8, y1 = (source.top + source.bottom) / 2, x2 = dest.left + 8, y2 = (dest.top + dest.bottom) / 2;
  graphNode(host, from).dispatchEvent(new PointerEvent('pointerdown', { bubbles: true, cancelable: true, pointerId: 12, pointerType: 'pen', button: 0, clientX: x1, clientY: y1 }));
  window.dispatchEvent(new PointerEvent('pointermove', { bubbles: true, cancelable: true, pointerId: 12, pointerType: 'pen', buttons: 1, clientX: (x1 + x2) / 2, clientY: (y1 + y2) / 2 }));

    assert(original.cardId && original.reference && reference.cardId === original.cardId && reference.reference, 'Local source and reference do not share identity');
    assert(store.cards.length === before.cards.length + 1, 'Expected exactly one canonical card');
    store.updateCard(original.cardId!, { title: '共享标题', front: '共享正文', back: '共享笔记' });
    for (const node of [original, reference]) { const content = nodeContent(store, node); equal([content.heading, content.excerpt, content.note], ['共享标题', '共享正文', '共享笔记'], 'Materialized node not synchronized'); }
    assert(store.undoLibraryChange(), 'Undo source edit'); assert(store.undoLibraryChange(), 'Undo reference materialization');
    equal(store.cards, before.cards, 'Undo materialization changed cards'); equal(store.nodes, before.nodes, 'Undo materialization changed original node'); equal(relationship(store), before.relations, 'Undo materialization changed relationships');
    return { source: original.id, reference: reference.id };
  });
  if (errors.length) throw new Error('Unhandled browser errors: ' + errors.join('; '));
} catch (error) { (window as any).fatal = String(error); }
