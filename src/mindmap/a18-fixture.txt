(MarkdownRenderer as any).render = async (_app: any, text: string, host: HTMLElement) => {
  const match = /\[([^\]]+)\]\(([^)]+)\)/.exec(text);
  if (!match) { host.textContent = text; return; }
  const link = document.createElement('a'); link.textContent = match[1]; link.href = match[2];
  host.append(document.createTextNode(text.slice(0, match.index)), link, document.createTextNode(text.slice(match.index + match[0].length)));
};

async function fresh() {
  let disk: any = {}; const files = new Map<string, string>();
  const app: any = { vault: { getFiles: () => [], getAbstractFileByPath: () => null, adapter: { exists: async (path: string) => files.has(path), mkdir: async (path: string) => files.set(path, ''), write: async (path: string, value: string) => files.set(path, value), read: async (path: string) => files.get(path) } } };
  const storage: any = { app, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } };
  let store = new Store(storage); await store.load(); stores.push(store); store.createStudySet('A18'); const canvasId = store.activeCanvasId!;
  store.addCard({ id: 'card-root', docPath: 'source.pdf', type: 'flashcard', title: '标题', front: '原文 [链接](https://example.com)', back: '答案正文', tags: ['样本'], links: [], createdAt: '2026-01-01T00:00:00.000Z' });
  store.addCard({ id: 'card-child', docPath: 'source.pdf', type: 'comment', title: '子卡', front: '子卡正文', back: '子卡笔记', tags: [], links: ['card-root'], createdAt: '2026-01-02T00:00:00.000Z' });
  store.addNode({ id: 'root', canvasId, parentId: null, cardId: 'card-root', title: '节点标题', collapsed: false, x: 40, y: 40 });
  for (let index = 1; index <= 10; index++) store.addNode({ id: `node-${String(index).padStart(2, '0')}`, canvasId, parentId: 'root', cardId: 'card-child', title: `分支 ${index}`, collapsed: false, x: 380, y: index * 150 });
  const host = document.createElement('section'); host.style.cssText = 'height:650px;width:1050px;max-width:100%;position:relative'; target.append(host);
  const p: any = { store, canvasId, host, storage };
  p.mount = () => { const plugin: any = { app, store: p.store, activeReader: null, viewEvents: {}, refreshSidebar() {}, refreshMindmap() {} }; p.view = new MindmapRenderer(plugin, { nativeChrome: true }); p.view.mount(host); mounted.push(p.view); };
  p.reload = async () => { await p.store.flush(); p.view.dispose(); mounted.splice(mounted.indexOf(p.view), 1); p.store = new Store(storage); stores.push(p.store); await p.store.load(); p.mount(); await wait(); };
  p.mount(); await wait(); return p;
}
