async function rejects(run: () => Promise<any>, pattern: RegExp) {
  try { await run(); } catch (error) { assert(pattern.test(String(error)), `Unexpected rejection ${error}`); return error as any; }
  throw new Error('Unexpected success');
}

async function fresh() {
  let disk: any = { settings: { backupEnabled: false } }; const entries = new Map<string, any>(); const text = new Map<string, string>(); const binary = new Map<string, ArrayBuffer>();
  let failWrite = false, failOpen = false, onBinary: (() => void) | undefined; const opened: string[] = [], deleted: string[] = [];
  const vault: any = {
    getAbstractFileByPath: (path: string) => entries.get(path), getFiles: () => [...entries.values()].filter(file => file instanceof TFile),
    createFolder: async (path: string) => { if (entries.has(path)) throw new Error('Already exists'); const folder = { path }; entries.set(path, folder); return folder; },
    create: async (path: string, value: string) => { if (failWrite && path.endsWith('.html')) throw new Error('Synthetic write denied'); if (entries.has(path)) throw new Error('Already exists'); const file = new TFile(path, value.length); entries.set(path, file); text.set(path, value); return file; },
    createBinary: async (path: string, value: ArrayBuffer) => { const file = new TFile(path, value.byteLength); entries.set(path, file); binary.set(path, value); onBinary?.(); return file; },
    delete: async (entry: any) => { deleted.push(entry.path); for (const path of entries.keys()) if (path === entry.path || path.startsWith(entry.path + '/')) { entries.delete(path); text.delete(path); binary.delete(path); } },
    adapter: { exists: async (path: string) => entries.has(path), mkdir: async (path: string) => { entries.set(path, { path }); }, read: async (path: string) => text.get(path) ?? '', write: async (path: string, value: string) => { text.set(path, value); } },
  };
  const app: any = { vault };
  const host: any = { app, loadData: async () => structuredClone(disk), saveData: async (data: any) => { disk = structuredClone(data); } };
  const store = new Store(host); await store.load(); stores.push(store); const set = store.createStudySet('Imported articles');
  const plugin: any = { app, store, openDocument: async (file: TFile) => { if (failOpen) throw new Error('Synthetic reader unavailable'); opened.push(file.path); return { getFilePath: () => file.path }; } };
  return { plugin, store, set, entries, text, binary, opened, deleted, host, failWrite: () => { failWrite = true; }, failOpen: () => { failOpen = true; }, onBinary: (callback: () => void) => { onBinary = callback; } };
}
