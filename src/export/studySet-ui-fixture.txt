const results: any[] = [];
(window as any).results = results;
(HTMLElement.prototype as any).empty = function () { this.replaceChildren(); };
const assert = (value: unknown, message: string) => { if (!value) throw new Error(message); };
const test = async (name: string, run: () => Promise<void>) => { try { await run(); results.push({name,passed:true}); } catch (error) { results.push({name,passed:false,error:String(error)}); } };
const button = (root: ParentNode, label: string) => [...root.querySelectorAll<HTMLButtonElement>('button')].find(item => item.getAttribute('aria-label') === label || item.textContent === label)!;

async function fixture() {
  const app: any = { vault: { getFiles: () => [], getAbstractFileByPath: () => null, adapter: { exists: async () => false } } };
  const store = new Store({app,loadData:async()=>({}),saveData:async()=>{}} as any); await store.load();
  const set = store.createStudySet('Fixture');
  const plugin: any = { store, app, refreshSidebar(){}, refreshMindmap(){} };
  const host = document.createElement('div'); host.style.height = '400px'; document.querySelector('#fixtures')!.append(host);
  const renderer = new MindmapRenderer(plugin,{nativeChrome:true}); renderer.mount(host);
  const open = () => { button(host,'选择脑图').click(); return button(document.querySelector('#test-menu')!,'导出学习集'); };
  return {store,set,renderer,host,open};
}

try {
  await test('nonempty study set enables export',async()=>{
    const {store,renderer,host,open}=await fixture();
    store.addCard({id:'source',docPath:'',type:'excerpt',front:'content',tags:[],links:[],createdAt:'2026-01-01T00:00:00Z'});
    assert(!open().disabled,'nonempty study set export is disabled');renderer.dispose();host.remove();
  });
  await test('empty study set disables export',async()=>{
    const {renderer,host,open}=await fixture();assert(open().disabled,'empty study set export is enabled');renderer.dispose();host.remove();
  });
  await test('document-only study set enables export',async()=>{
    const {store,set,renderer,host,open}=await fixture();store.updateStudySet(set.id,{docPaths:['document.pdf']});
    assert(!open().disabled,'document-only study set export is disabled');renderer.dispose();host.remove();
  });
  await test('local-node-only study set enables export',async()=>{
    const {store,renderer,host,open}=await fixture();store.addNode({id:'local',canvasId:store.activeCanvasId,title:'Local note',parentId:null,x:1,y:2,collapsed:false});
    assert(!open().disabled,'local-node-only study set export is disabled');renderer.dispose();host.remove();
  });
  await test('clicking export opens save dialog and publishes a readable ZIP',async()=>{
    const {store,renderer,host,open}=await fixture();
    store.addCard({id:'source',docPath:'',type:'excerpt',front:'content',tags:[],links:[],createdAt:'2026-01-01T00:00:00Z'});
    const before=JSON.stringify({cards:store.cards,nodes:store.nodes,edges:store.edges,studySets:store.studySets}),files=new Map<string,Uint8Array>();
    const previous=(window as any).require;let dialogs=0;
    (window as any).require=(name:string)=>name==='@electron/remote'?{dialog:{showSaveDialog:async()=>{dialogs++;return {canceled:false,filePath:'/virtual/result.marginstudy-set.zip'};}}}:name==='path'?{dirname:()=>'/virtual',join:(...parts:string[])=>parts.join('/')}:{promises:{writeFile:async(path:string,bytes:Uint8Array)=>{files.set(path,bytes.slice());},link:async(from:string,to:string)=>{files.set(to,files.get(from)!);},unlink:async(path:string)=>{files.delete(path);}}};
    try{
      const item=open();assert(!item.disabled,'nonempty item disabled');item.click();
      const modal=document.querySelector<HTMLElement>('.ms-modal')!, checkbox=modal.querySelector<HTMLInputElement>('[aria-label="历史版本"]')!, select=modal.querySelector<HTMLSelectElement>('[aria-label="历史版本范围"]')!;
      assert(checkbox.checked&&select.options.length===2&&select.options[0].textContent==='最近 5 个'&&select.options[1].textContent==='全部版本','history options missing');
      checkbox.checked=false;checkbox.dispatchEvent(new Event('change'));assert(select.disabled,'history range not disabled');checkbox.checked=true;checkbox.dispatchEvent(new Event('change'));assert(!select.disabled,'history range not enabled');
      button(modal,'导出').click();
      for(let attempt=0;attempt<100&&!files.has('/virtual/result.marginstudy-set.zip');attempt++)await new Promise(done=>setTimeout(done,10));
      assert(dialogs===1,'save dialog was not opened');const bytes=files.get('/virtual/result.marginstudy-set.zip');assert(bytes,'ZIP was not published');
      const payload=JSON.parse(strFromU8(unzipSync(bytes! as Uint8Array)['studyset.json']));
      assert(JSON.stringify({cards:payload.cards,nodes:payload.nodes,edges:payload.edges,studySets:payload.studySets})===before,'ZIP does not match current store');
      assert(store.getLibraryHistoryState().undoLabel==='导出学习集','export not in history');assert(store.undoLibraryChange(),'export not undoable');
      assert(JSON.stringify({cards:store.cards,nodes:store.nodes,edges:store.edges,studySets:store.studySets})===before,'undo changed business objects');
    }finally{(window as any).require=previous;renderer.dispose();host.remove();}
  });
} finally { (window as any).done = true; }
