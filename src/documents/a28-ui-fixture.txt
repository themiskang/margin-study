import { Store } from "__PROJECT__/src/model/store";
import { openSyncStatusPanel } from "__PROJECT__/src/documents/recoveryPrompts";

const results: { name: string; passed: boolean; error?: string }[] = [];
const check = async (name: string, action: () => void | Promise<void>) => {
  try { await action(); results.push({ name, passed: true }); }
  catch (error) { results.push({ name, passed: false, error: String((error as Error)?.stack ?? error) }); }
};
const assert = (condition: unknown, message: string) => { if (!condition) throw new Error(message); };
const click = (root: ParentNode, text: string) => {
  const button = [...root.querySelectorAll("button")].find(item => item.textContent?.trim() === text);
  if (!button) throw new Error("button missing: " + text);
  button.click();
};
const wait = async (predicate: () => unknown) => {
  for (let index = 0; index < 100; index++) { if (predicate()) return; await new Promise(resolve => setTimeout(resolve, 10)); }
  throw new Error("condition did not become true");
};
const stamp = "2020-01-01T00:00:00.000Z";
let disk: any = { revision: 1, cards: [
  { id: "one", type: "excerpt", docPath: "fixture.pdf", front: "原文一", tags: [], links: [], createdAt: stamp },
  { id: "two", type: "excerpt", docPath: "fixture.pdf", front: "原文二", tags: [], links: [], createdAt: stamp },
  { id: "three", type: "excerpt", docPath: "fixture.pdf", front: "待删除", tags: [], links: [], createdAt: stamp },
], nodes: [], edges: [], srs: {}, documents: [{ id: "doc", path: "fixture.pdf", kind: "pdf", createdAt: stamp }], studySets: [], canvases: [] };
const files = new Map<string, string>(), dirs = new Set<string>();
const adapter = { exists: async (path: string) => files.has(path) || dirs.has(path), mkdir: async (path: string) => { dirs.add(path); },
  read: async (path: string) => { const value = files.get(path); if (value === undefined) throw new Error("missing fixture file"); return value; },
  write: async (path: string, content: string) => { files.set(path, content); },
  list: async (path: string) => ({ files: [...files.keys()].filter(value => value.startsWith(path + "/")), folders: [] }) };
const store = new Store({ app: { vault: { adapter, getFiles: () => [] } } as any, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } });
let sourceCalls = 0, conflictCalls = 0; const notices: string[] = [];
try {

  const external = structuredClone(disk); external.cards[0].front = "外部版本"; disk = external;

    assert(rows.length === 7, "status row count");
    for (const row of rows) {

  });

    assert(modal.querySelector('[data-status="pending-upload"]')?.textContent?.includes("等待上传"), "pending-upload text");
    assert(modal.querySelector('[data-status="missing"]')?.textContent?.includes("iCloud"), "missing row does not distinguish cloud availability");
    assert(modal.querySelector('[data-status="deleted"]')?.textContent?.includes("对象已删除"), "deleted text");

    assert(!![...modal.querySelectorAll("button")].find(item => item.textContent === "立即保存"), "save action missing");
  });
  await store.resolveDataConflict("merge");
  store.updateCard("one", { front: "恢复前修改" }); store.updateCard("two", { front: "保留的后续修改" }); await store.flush();

  });

    store.updateCard("two", { front: "再次编辑" }); await Promise.resolve();

  });
