import { ReaderView } from '__PROJECT__/src/pdf/readerView';

export async function runDocumentNotebookAcceptance({test,assert,wait,plugin,add,app,htmlController,selection}:any){
 await test('Document notebooks isolate all annotations and handwriting layers, preserve legacy IDs, and source jumps switch notebooks',async()=>{
  const p=plugin(),a=htmlController(p,'notebooks.html'),b=htmlController(p,'notebooks.html');await a.load();await b.load();assert(!a.ann.createDocumentNotebook('空白新本'),'empty notebook allowed redundant creation');
  a.ann.commitAnnotation(a.ann.resolveSelection(selection(a.body,'target')),'highlight','#ff0','target');a.ann.createInkLayer('主本手写');const originalLayer=a.ann.getInkLayers().active;a.ann.createPenAnnotation([[{x:.1,y:.1},{x:.2,y:.2}]]);
  const annotations=await p.store.loadAnnotations('notebooks.html'),legacy=annotations[0];delete legacy.documentNotebookId;
  assert(a.ann.createDocumentNotebook('错题复盘'),'second notebook not created');const other=a.ann.getDocumentNotebookId();assert(other!=='default'&&!a.body.querySelector('[data-ann-id]')&&!b.body.querySelector('[data-ann-id]'),'old annotations leaked into new notebook');assert(a.ann.getInkLayers().layers.length===1,'source handwriting layers leaked');
  a.ann.commitAnnotation(a.ann.resolveSelection(selection(a.body,'target')),'underline','#f00','target');a.ann.createInkLayer('二轮专用');const ownLayer=a.ann.getInkLayers().active;a.ann.createPenAnnotation([[{x:.4,y:.4},{x:.5,y:.5}]]);const otherAnn=annotations[2];
  a.ann.selectDocumentNotebook('default');assert(a.ann.getInkLayers().active===originalLayer&&!a.ann.getInkLayers().layers.some((layer:any)=>layer.id===ownLayer),'per-notebook handwriting state not restored');assert(a.body.querySelectorAll('[data-ann-id]').length===2,'wrong active notebook annotation count');
  a.ann.setDocumentNotebookOverlay(other);const refs=a.body.querySelectorAll('.ms-document-notebook-reference');assert(refs.length===2&&[...refs].every((element:any)=>getComputedStyle(element).opacity==='0.35'&&getComputedStyle(element).pointerEvents==='none'),'notebook overlay editable or not visible');
  const referenceRange=a.body.querySelector('.ms-html-reference-range')!,activeText=a.body.querySelector('.ms-hl')!;assert(referenceRange&&referenceRange.querySelector('.ms-html-reference-segment')&&activeText.textContent==='target'&&!referenceRange.contains(activeText),'overlapping reference range lost or wrapped active text');
  const beforeWidth=referenceRange.firstElementChild!.getBoundingClientRect().width;a.body.style.fontSize='26px';await wait(80);const afterWidth=referenceRange.firstElementChild!.getBoundingClientRect().width;assert(afterWidth>beforeWidth&&Math.abs(afterWidth-activeText.getBoundingClientRect().width)<1,'reference did not track text reflow');assert(a.body.textContent==='before target after','reference duplicated source text');
  a.ann.setRecallMode(true);assert([...refs].every((element:any)=>getComputedStyle(element).display==='none')&&getComputedStyle(activeText).display!=='none'&&a.body.textContent==='before target after','notebook reference leaked into recall or hid source text');a.ann.setRecallMode(false);a.ann.setDocumentNotebookOverlay(undefined);
  a.ann.flashAnnotation(otherAnn.id);assert(a.ann.getDocumentNotebookId()===other&&a.ann.getInkLayers().active===ownLayer,'source jump failed to activate owning notebook');assert(p.store.cards[0].anchorAnnId===legacy.id&&legacy.id===annotations[0].id,'switch rewrote legacy source ID');
  await a.close();await b.close();assert(p.store.subscriptionCount===0&&p.store.annotationSubscriptionCount===0,'notebook views leaked subscriptions');
 });
 await test('PDF notebooks isolate folds/bookmarks, share layout and keep a pending stroke in its original notebook',async()=>{
  add('notebooks.pdf',{name:'Notebooks',pages:3});const p=plugin(),v:any=new ReaderView({app} as any,p);await v.onOpen();await v.loadFileByPath('notebooks.pdf');const c=v.annController;
  c.commitAnnotation({kind:'pdf',pageNum:1,rects:[{x:20,y:500,w:100,h:18}]},'highlight','#ff0','before');v.toggleFold(2);v.toggleBookmark(1);await v.setPageLayout('spread',false);await v.renderPage(1);c.selectPenTool('draw');
  const wrapper=v.getPageRecord(1).wrapper,overlay=wrapper.querySelector('.ms-pen-overlay'),bounds=wrapper.getBoundingClientRect();overlay.dispatchEvent(new PointerEvent('pointerdown',{pointerId:17,clientX:bounds.left+60,clientY:bounds.top+130,button:0,isPrimary:true,bubbles:true}));overlay.dispatchEvent(new PointerEvent('pointermove',{pointerId:17,clientX:bounds.left+100,clientY:bounds.top+140,button:0,isPrimary:true,bubbles:true}));
  assert(v.createDocumentNotebook('二轮'),'new PDF notebook not created');const second=v.getDocumentNotebookId(),annotations=await p.store.loadAnnotations('notebooks.pdf');assert(annotations.find((ann:any)=>ann.type==='pen').documentNotebookId==='default','pending ink migrated to new notebook');assert(!v.folded.size&&!v.bookmarked.size&&v.getPageLayout().layout==='spread'&&!v.getPageLayout().continuous,'new notebook did not preserve global layout or isolate metadata');
  c.createInkLayer('重做');c.createPenAnnotation(1,[[{x:.5,y:.5},{x:.6,y:.5}]]);const source=annotations[annotations.length-1];
  await v.selectDocumentNotebook('default');assert(v.folded.has(2)&&v.bookmarked.has(1)&&v.getPageLayout().layout==='spread'&&!v.getPageLayout().continuous,'original notebook view metadata lost');
  v.flashAnnotation(source.id);await wait(100);const canvas=v.getPageRecord(1).wrapper.querySelector('.ms-ink-raster') as HTMLCanvasElement;assert(v.getDocumentNotebookId()===second&&!v.folded.has(2)&&canvas?.getContext('2d')!.getImageData(Math.round(canvas.width*.55),Math.round(canvas.height*.5),1,1).data[3]>0,'PDF source jump wrong notebook');await v.onClose();
  const reopened:any=new ReaderView({app} as any,p);await reopened.onOpen();await reopened.loadFileByPath('notebooks.pdf');assert(reopened.getDocumentNotebookId()===second&&reopened.annController.getInkLayers().layers.some((layer:any)=>layer.name==='重做'),'notebook state not restored');await reopened.onClose();
 });
 await test('Document notebook manager creates/renames and distinguishes clearing current from deleting an inactive notebook',async()=>{
  add('notebook-manager.pdf',{name:'Notebook manager',pages:1});const p=plugin(),v:any=new ReaderView({app} as any,p);await v.onOpen();await v.loadFileByPath('notebook-manager.pdf');const c=v.annController;c.createPenAnnotation(1,[[{x:.1,y:.1}]]);v.openDocumentNotebooks();
  const button=(root:ParentNode,text:string)=>[...root.querySelectorAll('button')].find((element)=>element.textContent===text)! as HTMLButtonElement;
  const modal=document.querySelector('.ms-document-notebook-modal')!;button(modal,'+ 新建文档笔记本').click();let editor=document.querySelector('.ms-document-notebook-name-modal')!,name=editor.querySelector('input')!;name.value='专题';button(editor,'确定').click();const second=v.getDocumentNotebookId();assert(second!=='default','notebook UI create failed');c.createPenAnnotation(1,[[{x:.2,y:.2}]]);
  (modal.querySelector('[aria-label="编辑文档笔记本"]') as HTMLButtonElement).click();(modal.querySelector('[aria-label="重命名笔记本 专题"]') as HTMLButtonElement).click();editor=document.querySelector('.ms-document-notebook-name-modal')!;name=editor.querySelector('input')!;name.value='复盘';button(editor,'确定').click();assert(c.getDocumentNotebooks().notebooks.some((item:any)=>item.name==='复盘'),'rename failed');
  (modal.querySelector('[aria-label="删除笔记本 复盘"]') as HTMLButtonElement).click();button(document.querySelector('.ms-document-notebook-delete-modal')!,'清空').click();assert(c.getDocumentNotebooks().notebooks.some((item:any)=>item.id===second)&&!(await p.store.loadAnnotations('notebook-manager.pdf')).some((ann:any)=>ann.documentNotebookId===second),'clearing current removed notebook or retained notes');
  (modal.querySelector('[aria-label="切换笔记本 主笔记本"]') as HTMLButtonElement).click();(modal.querySelector('[aria-label="删除笔记本 复盘"]') as HTMLButtonElement).click();button(document.querySelector('.ms-document-notebook-delete-modal')!,'删除').click();assert(!c.getDocumentNotebooks().notebooks.some((item:any)=>item.id===second),'inactive notebook not removed');await v.onClose();assert(!document.querySelector('.ms-document-notebook-modal'),'notebook manager leaked on close');
 });
}
