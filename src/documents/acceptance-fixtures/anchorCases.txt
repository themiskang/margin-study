export function croppedPdf(rotations=[0]): ArrayBuffer {
  const stream='BT /F1 12 Tf 100 600 Td (Synthetic CropBox reference) Tj ET\n';
  const objects=['<< /Type /Catalog /Pages 2 0 R >>',`<< /Type /Pages /Kids [${rotations.map((_,index)=>`${index+5} 0 R`).join(' ')}] /Count ${rotations.length} >>`,
    '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>',`<< /Length ${stream.length} >>\nstream\n${stream}endstream`,
    ...rotations.map(rotation=>`<< /Type /Page /Parent 2 0 R /MediaBox [0 0 600 800] /CropBox [50 70 550 770] /Rotate ${rotation} /Resources << /Font << /F1 3 0 R >> >> /Contents 4 0 R >>`)];
  let source='%PDF-1.4\n'; const offsets=[0];
  objects.forEach((object,index)=>{offsets.push(source.length);source+=`${index+1} 0 obj\n${object}\nendobj\n`;});
  const xref=source.length;source+=`xref\n0 ${objects.length+1}\n0000000000 65535 f \n`;
  source+=offsets.slice(1).map(offset=>`${String(offset).padStart(10,'0')} 00000 n \n`).join('');
  source+=`trailer\n<< /Size ${objects.length+1} /Root 1 0 R >>\nstartxref\n${xref}\n%%EOF\n`;
  return new TextEncoder().encode(source).buffer as ArrayBuffer;
}
