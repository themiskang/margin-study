class VoiceEngine {
 spoken:any[]=[];current:any=null;pauses=0;resumes=0;cancels=0;
 create(text:string){return {text,rate:1,lang:'',onend:null,onerror:null,onboundary:null} as any;}
 speak(utterance:any){this.current=utterance;this.spoken.push(utterance);}
 pause(){this.pauses++;}resume(){this.resumes++;}cancel(){this.cancels++;this.current=null;}
 finish(){const utterance=this.current;this.current=null;utterance?.onend?.({});}
}
export async function runSpeechAcceptance({test,assert,wait,plugin,add,app,selection}:any){
 await test('Shared speech continuously queues long text, supports pause/resume, live rate and completion',async()=>{
  const engine=new VoiceEngine(),session=new SpeechCoordinator(()=>engine).createSession('long');
  const text='完整句子用于连续阅读。'.repeat(450);assert(await session.start(text),'speech did not start');
  assert(engine.spoken[0].text.length<=220,'long utterance not chunked');session.pause();assert(session.state.status==='paused'&&engine.pauses===1,'pause failed');
  const before=engine.spoken.length;engine.finish();await wait();assert(engine.spoken.length===before,'paused queue continued');
  session.resume();await wait();assert(engine.spoken.length===before+1,'resume did not continue');
  const current=engine.current;current.onboundary({charIndex:8});session.setRate(1.75);await wait();
  assert(engine.current.rate===1.75&&engine.current.text===current.text.slice(8).trimStart(),'live rate lost boundary');
  for(let i=0;i<100&&session.state.status!=='finished';i++){engine.finish();await wait();}
  assert(session.state.status==='finished'&&engine.spoken.length>20,'full document was truncated');
  assert(speechChunks('😀'.repeat(30),7).every(part=>!/^[\uDC00-\uDFFF]|[\uD800-\uDBFF]$/.test(part)),'chunk split emoji');session.dispose();
 });
 await test('Speech stop cancels pending sources immediately; owners cannot cancel another reader',async()=>{
  const engine=new VoiceEngine(),coordinator=new SpeechCoordinator(()=>engine),a=coordinator.createSession('a'),b=coordinator.createSession('b');
  const pending=a.start(async function*(){await new Promise(()=>{});yield{text:'stale'};});await wait();a.stop();
  const result=await Promise.race([pending,wait(80).then(()=> 'timeout')]);assert(result===false,'stop waited for uncooperative source');
  await a.start('reader a');await b.start('reader b');assert(a.state.status==='stopped'&&b.state.status==='speaking','ownership switch failed');
  const cancellations=engine.cancels;a.dispose();assert(engine.cancels===cancellations&&engine.current.text==='reader b','unrelated view cancelled owner');
  b.pause();b.setRate(2);await wait();assert(b.state.status==='paused'&&!engine.current,'paused rate restarted early');b.resume();await wait();assert(engine.current.rate===2,'paused rate not applied');
  b.dispose();const unavailable=new SpeechCoordinator(()=>null).createSession('unsupported');assert(!await unavailable.start('text')&&unavailable.state.status==='error','unsupported engine falsely succeeded');unavailable.dispose();
  const empty=coordinator.createSession('empty');assert(!await empty.start('  ')&&empty.state.status==='error','empty source falsely finished');
  let closed=false;const source={async *[Symbol.asyncIterator](){try{yield{text:'error case'};yield{text:'must not continue'};}finally{closed=true;}}};
  await empty.start(source);engine.current.onerror({error:'voice-unavailable'});await wait();assert(empty.state.status==='error'&&empty.state.error==='voice-unavailable'&&closed,'voice error not reported or source not closed');empty.dispose();
 });

   for(let i=0;i<3;i++){assert(engine.current.text.includes('Spoken page '+(i+1)),'PDF page order');assert(pdf.getState().position.page===i+1,'spoken page did not follow source position');engine.finish();await wait(35);}
   assert(pdf.speech.session.state.status==='finished','PDF did not finish');
   await word.loadFileByPath('speech.docx');assert(await word.speech.start()&&engine.current.text==='Word spoken body','Word speech missing');

  }finally{await pdf.onClose();if(!html.destroyed)await html.onClose();if(!word.destroyed)await word.onClose();
   if(speechDescriptor)Object.defineProperty(window,'speechSynthesis',speechDescriptor);else delete(window as any).speechSynthesis;

  }
