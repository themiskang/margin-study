function pointer(host:HTMLElement,type:string,x:number,y:number){const overlay=host.querySelector('.ms-pen-overlay')!,rect=host.getBoundingClientRect();overlay.dispatchEvent(new PointerEvent(type,{clientX:rect.left+x*rect.width,clientY:rect.top+y*rect.height,pointerId:11,button:0,isPrimary:true,bubbles:true}));}

 await test('Handwriting layers preserve legacy ink/non-ink, switch and overlay one read-only layer, hide reference in recall',async()=>{

  a.ann.createPenAnnotation([[{x:.2,y:.2},{x:.4,y:.2}]]);const current=(await p.store.loadAnnotations('layers.html')).find((ann:any)=>ann.type==='pen'&&ann.id!==legacy.id);
