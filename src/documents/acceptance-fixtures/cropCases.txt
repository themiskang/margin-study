export async function runCropAcceptance(context:any) {
  const {test,assert,wait,plugin,add,app,realPdf}=context;
  const near=(actual:number,expected:number,message:string)=>assert(Math.abs(actual-expected)<1.2,`${message}: ${actual} vs ${expected}`);
  await test('Real PDF selection folds cropped and rotated source pixels without rewriting source or excerpts',async()=>{
    await realPdf.initPdfWorker(app);const path='real-region-fold.pdf',source=croppedPdf([0,90,180,270]);add(path,{bytes:source});const p=plugin(),view=new RealReaderView({app} as any,p as any);await view.onOpen();

    try{
      await view.loadFileByPath(path);view.setScale(1);const sourceHash=Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',source))).join(',');
      for(let page=1;page<=4;page++){

        rec=view.getPageRecord(page)!;assert(rec.canvas.height<before-3,'real PDF pixels were not contracted at rotation '+(page-1)*90);assert([...rec.textLayer.querySelectorAll('span')].every(item=>getComputedStyle(item).display==='none'),'real PDF folded text remains selectable');const foldedBytes=await app.vault.readBinary(app.vault.getAbstractFileByPath(path));assert(Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',foldedBytes))).join(',')===sourceHash,'fold changed source SHA-256');
        const pixels=rec.canvas.getContext('2d')!.getImageData(0,0,rec.canvas.width,rec.canvas.height).data;let dark=0;for(let i=0;i<pixels.length;i+=4)if(pixels[i]<150&&pixels[i+1]<150&&pixels[i+2]<150&&pixels[i+3]>0)dark++;assert(dark===0,'fold left source text pixels visible at rotation '+(page-1)*90+': '+dark);
        await view.undoLastEdit();await until(()=>!view.getPageRecord(page)!.wrapper.querySelector('[aria-label="展开折叠区域"]'));assert(view.getPageRecord(page)!.canvas.height===before,'real PDF fold undo changed page size');
      }
      assert(!(await p.store.loadAnnotations(path)).length&&!p.store.cards.length,'real fold created excerpts/cards');const bytes=new Uint8Array(await app.vault.readBinary(app.vault.getAbstractFileByPath(path)));assert(bytes.length===source.byteLength&&bytes.every((value,index)=>value===new Uint8Array(source)[index]),'real fold changed source bytes');const beforeHash=await crypto.subtle.digest('SHA-256',source),afterHash=await crypto.subtle.digest('SHA-256',bytes);assert(Array.from(new Uint8Array(beforeHash)).join(',')===Array.from(new Uint8Array(afterHash)).join(','),'real fold changed source SHA-256');assert(!p.store.getDocMeta(path).pdfRegionFolds,'real fold undo left a migration field');
    }finally{window.getSelection()?.removeAllRanges();await view.onClose();view.containerEl.remove();}
  });
  await test('Inserting a selection into a TOC preserves existing hierarchy and is undoable',async()=>{
    await realPdf.initPdfWorker(app);const path='selection-toc-hierarchy.pdf';add(path,{bytes:croppedPdf([0,0,0])});const p=plugin(),view=new RealReaderView({app} as any,p as any);await view.onOpen();

    }finally{window.getSelection()?.removeAllRanges();await view.onClose();view.containerEl.remove();}
  });
  await test('PDF crop applies to every rotated page, preserves text and both anchor formats, and survives reader reopen without rewriting source or annotations',async()=>{
    await realPdf.initPdfWorker(app);
    const path='crop-four-directions.pdf',source=croppedPdf([0,90,180,270]);add(path,{bytes:source});
    const p=plugin();
    const annotations=[1,2,3,4].flatMap(page=>[
      {id:'crop-canonical-'+page,docPath:path,docKind:'pdf',page,type:'highlight',rects:[{x:100,y:596,w:145,h:15}],pdfCoordinateSpace:'pdf',text:'source anchor',color:'#ffff00',createdAt:'2026-10-17'},
      {id:'crop-legacy-'+page,docPath:path,docKind:'pdf',page,type:'region',rects:[{x:100,y:200,w:60,h:35}],text:'legacy anchor',color:'#ff0000',createdAt:'2026-10-17'},
      {id:'crop-voice-'+page,docPath:path,docKind:'pdf',page,type:'voice',rects:[],pos:{x:.4,y:.4},audioPath:'synthetic-audio.webm',text:'',color:'#ff0000',createdAt:'2026-10-17'}
    ]);
    p.store.saveAnnotationsDebounced(path,annotations);await p.store.flush();const saved=JSON.stringify(await p.store.loadAnnotations(path));
    let view=new RealReaderView({app} as any,p as any);await view.onOpen();
    const baseline:any[]=[];
    try {
      await view.loadFileByPath(path);view.setScale(1);
      for(let page=1;page<=4;page++){
        await view.renderPage(page);const rec=view.getPageRecord(page)!;
        const text=rec.textLayer.querySelector('span')!.getBoundingClientRect(),canvas=rec.canvas.getBoundingClientRect();
        const highlight=rec.annLayer.querySelector('.ms-ann-highlight')!.getBoundingClientRect(),legacy=rec.annLayer.querySelector('.ms-ann-region')!.getBoundingClientRect();
        baseline.push({w:canvas.width,h:canvas.height,tx:text.left-canvas.left,ty:text.top-canvas.top,hx:highlight.left-canvas.left,hy:highlight.top-canvas.top,lx:legacy.left-canvas.left,ly:legacy.top-canvas.top});
      }
      await view.setPageCrop({left:.05,top:.05,right:.95,bottom:.95});
      for(let page=1;page<=4;page++){
        await view.renderPage(page);const rec=view.getPageRecord(page)!,before=baseline[page-1],canvas=rec.canvas.getBoundingClientRect();
        near(canvas.width,page%2===1?450:630,'cropped paper width');near(canvas.height,page%2===1?630:450,'cropped paper height');
        const text=rec.textLayer.querySelector('span')!.getBoundingClientRect(),highlight=rec.annLayer.querySelector('.ms-ann-highlight')!.getBoundingClientRect(),legacy=rec.annLayer.querySelector('.ms-ann-region')!.getBoundingClientRect();
        near(text.left-canvas.left,before.tx-before.w*.05,'text X page '+page+' '+JSON.stringify(before));near(text.top-canvas.top,before.ty-before.h*.05,'text Y page '+page);
        near(highlight.left-canvas.left,before.hx-before.w*.05,'canonical X');near(highlight.top-canvas.top,before.hy-before.h*.05,'canonical Y');
        near(legacy.left-canvas.left,before.lx-before.w*.05,'legacy X');near(legacy.top-canvas.top,before.ly-before.h*.05,'legacy Y');
        const voice=rec.annLayer.querySelector('.ms-ann-voice') as HTMLElement;
        near(parseFloat(voice.style.left),before.w*.35,'voice X');near(parseFloat(voice.style.top),before.h*.35,'voice Y');
      }
      assert(JSON.stringify(await p.store.loadAnnotations(path))===saved,'crop rewrote annotation sidecar');
      assert(Array.from(new Uint8Array(await app.vault.readBinary(app.vault.getAbstractFileByPath(path)))).join(',')===Array.from(new Uint8Array(source)).join(','),'source bytes changed');
      await view.onClose();view.containerEl.remove();view=new RealReaderView({app} as any,p as any);await view.onOpen();await view.loadFileByPath(path);view.setScale(1);await view.renderPage(1);
      assert(JSON.stringify(view.getPageCrop())===JSON.stringify({left:.05,top:.05,right:.95,bottom:.95}),'crop lost on reopen');
      near(view.getPageRecord(1)!.canvas.clientWidth,450,'reopened cropped width');
      assert(JSON.stringify(await p.store.loadAnnotations(path))===saved,'reopen rewrote annotations');
    } finally {await view.onClose();view.containerEl.remove();}
  });
  await test('PDF crop keeps ink in its original page basis while erasing, undoing and drawing in an extended margin',async()=>{
    const path='crop-ink.pdf';add(path,{bytes:croppedPdf()});const p=plugin();
    const oldInk={id:'old-crop-ink',docPath:path,docKind:'pdf',page:1,type:'pen',rects:[],text:'',color:'#ff0000',createdAt:'2026-10-17',strokes:[[{x:.25,y:.3,pressure:.2},{x:.45,y:.3,pressure:.8}],[{x:.6,y:.6},{x:.8,y:.6}]]};
    const before=JSON.stringify(oldInk);p.store.saveAnnotationsDebounced(path,[oldInk]);await p.store.flush();
    const view=new RealReaderView({app} as any,p as any);await view.onOpen();
    const draw=(points:number[][])=>{
      const host=view.getPageRecord(1)!.wrapper,overlay=host.querySelector('.ms-pen-overlay')!,bounds=host.getBoundingClientRect();
      const event=(type:string,x:number,y:number)=>overlay.dispatchEvent(new PointerEvent(type,{clientX:bounds.left+x*bounds.width,clientY:bounds.top+y*bounds.height,pointerId:1,button:0,isPrimary:true,bubbles:true}));
      event('pointerdown',points[0][0],points[0][1]);for(const [x,y] of points.slice(1))event('pointermove',x,y);event('pointerup',points.at(-1)![0],points.at(-1)![1]);
    };
    try {
      await view.loadFileByPath(path);await view.setPageCrop({left:.1,top:.1,right:.9,bottom:.9});await view.renderPage(1);
      const canvas=view.getPageRecord(1)!.wrapper.querySelector<HTMLCanvasElement>('canvas.ms-ink-raster')!,ctx=canvas.getContext('2d')!,sx=canvas.width/view.getPageRecord(1)!.annLayer.clientWidth,sy=canvas.height/view.getPageRecord(1)!.annLayer.clientHeight;
      assert(ctx.getImageData(Math.round(75*sx),Math.round(140*sy),1,1).data[3]>0&&ctx.getImageData(Math.round(60*sx),Math.round(140*sy),1,1).data[3]===0,'old ink did not start at crop-projected (75,140)');
      assert(JSON.stringify((await p.store.loadAnnotations(path))[0])===before,'crop rewrote old ink');
      view.setHandwritingMode('eraser');draw([[.3,.2],[.3,.3]]);
      assert(oldInk.strokes.length===1&&JSON.stringify(oldInk.strokes[0])===JSON.stringify([{x:.6,y:.6},{x:.8,y:.6}]),'erase missed projected ink or rewrote untouched stroke');
      view.undoLastEdit();assert(JSON.stringify(oldInk)===before,'ink undo lost original pressure/coordinates');view.redoLastEdit();assert(oldInk.strokes.length===1,'projected ink redo');view.undoLastEdit();
      await view.setPageCrop({left:-.2,top:-.1,right:1.2,bottom:1.1});view.setHandwritingMode('pen');draw([[.05,.2],[.1,.25]]);view.setHandwritingMode('off');
      const added=(await p.store.loadAnnotations(path)).find((ann:any)=>ann.id!==oldInk.id);assert(added,'extended margin ink was not saved');
      assert(Math.abs(added.strokes[0][0].x+.13)<.000001&&Math.abs(added.strokes[0][0].y-.14)<.000001,'extended ink original coordinates');
      assert(added.strokes[0][0].x<0,'margin point clamped into original paper');
      const addedBefore=JSON.stringify(added);await view.setPageCrop({left:0,top:0,right:1,bottom:1});view.undoLastEdit();
      assert((await p.store.loadAnnotations(path)).length===1,'crop change destroyed ink undo history');view.redoLastEdit();
      assert(JSON.stringify((await p.store.loadAnnotations(path)).find((ann:any)=>ann.id===added.id))===addedBefore,'redo changed extended stroke identity');
    }finally{await view.onClose();view.containerEl.remove();}
  });
