const control=(root:ParentNode,label:string)=>root.querySelector(`[aria-label="${label}"]`) as HTMLElement;

function pointer(target:EventTarget,type:string,x:number,y:number,id=71){target.dispatchEvent(new PointerEvent(type,{pointerId:id,isPrimary:true,button:0,clientX:x,clientY:y,bubbles:true,cancelable:true}));}
function overlayControls(element:HTMLElement){return (element.closest('.ms-page')?.querySelector(`.ms-free-overlay-chrome[data-ann-id="${element.dataset.annId}"]`) as HTMLElement|null)??element;}
function startDrag(element:HTMLElement,resize=false){const target=control(overlayControls(element),resize?'缩放批注':'移动批注'),box=element.getBoundingClientRect();pointer(target,'pointerdown',box.left+12,box.top+12);return{x:box.left+12,y:box.top+12};}
function drag(element:HTMLElement,dx:number,dy:number,resize=false){const at=startDrag(element,resize);pointer(window,'pointermove',at.x+dx,at.y+dy);pointer(window,'pointerup',at.x+dx,at.y+dy);}
async function imageFile(color='#397bdf'){

 const blob=await new Promise<Blob>(resolve=>canvas.toBlob(value=>resolve(value!),'image/png'));return new File([blob],'synthetic-image.png',{type:'image/png'});
}
