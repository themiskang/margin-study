function pointer(overlay:SVGSVGElement,host:HTMLElement,type:string,x:number,y:number,extra:any={}){
 const bounds=host.getBoundingClientRect();overlay.dispatchEvent(new PointerEvent(type,{clientX:bounds.left+x*bounds.width,clientY:bounds.top+y*bounds.height,pointerId:1,button:0,isPrimary:true,bubbles:true,...extra}));
}
function drag(overlay:SVGSVGElement,host:HTMLElement,points:number[][]){pointer(overlay,host,'pointerdown',...points[0] as [number,number]);for(const [x,y] of points.slice(1))pointer(overlay,host,'pointermove',x,y);pointer(overlay,host,'pointerup',...points[points.length-1] as [number,number]);}
