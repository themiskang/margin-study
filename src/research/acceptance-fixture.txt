async function run(){
await test('Eight modes and empty search make no implicit network navigation',()=>{ok(control('研究搜索类型').options.length===8,'eight modes');ok(history.length===0,'implicit navigation');ok(control('将网页转为 PDF').disabled,'premature export');});
await test('Saved malformed engines are discarded, missing modes restored, selected engine repaired',()=>{const p=researchPreferences({order:['pdf','pdf'] as any,mode:'unknown' as any,selected:{text:'absent'},engines:[{id:'bad',mode:'text',name:'bad',url:'javascript:%s'}] as any});ok(p.order.length===8&&p.order[0]==='pdf'&&p.mode==='video','normalization');ok(p.selected.text==='bing'&&!p.engines.some(e=>e.id==='bad'),'unsafe engine');});
await test('Engine substitution encodes Chinese, separators and every placeholder',()=>{const url=engineSearchUrl({id:'custom',mode:'text',name:'x',url:'https://example.test/?q=%s&mirror=%s'},'中文 & ?/');ok(new URL(url).searchParams.get('q')==='中文 & ?/'&&new URL(url).searchParams.get('mirror')==='中文 & ?/','query escaping');});
await test('Unsafe URL schemes and embedded credentials are rejected',()=>{for(const input of ['javascript:alert(1)','file:///tmp/a','https://name:secret@example.test/']){let rejected=false;try{researchUrl(input);}catch{rejected=true;}ok(rejected,'allowed '+input);}});
await test('Manual text search and mode switching navigate the actual browser port',async()=>{change('研究搜索类型','text');control('研究搜索内容').value='Synthetic learning';control('开始研究搜索').dispatchEvent(new Event('submit',{bubbles:true,cancelable:true}));await wait();ok(new URL(history[index]).searchParams.get('q')==='Synthetic learning','search not passed');change('研究搜索类型','scholar');await wait();ok(history[index].includes('scholar.google.com'),'wrong category');});
await test('Back and forward preserve the browsing session',()=>{click('研究网页后退');ok(index===0&&!control('研究网页前进').disabled,'back');click('研究网页前进');ok(index===1&&!control('研究网页后退').disabled,'forward');});
await test('Lock rejects new selections; unlock follows new selection; old selection is ignored',async()=>{const before=history.length;click('锁定研究内容');handle.setSelection('locked synthetic');ok(history.length===before,'locked changed');click('锁定研究内容');handle.setSelection('expired synthetic',Date.now()-6000);ok(history.length===before,'expired changed');handle.setSelection('Fresh synthetic');await wait();ok(history.length===before+1&&decodeURIComponent(history[index]).includes('Fresh synthetic'),'unlocked selection');});
await test('Eight category choices and custom engine selection persist without losing the page',async()=>{for(const id of ['video','text','image','scholar','question','dictionary','pdf','translate']){change('研究搜索类型',id);await wait();ok(plugin.store.settings.research.mode===id&&history[index].startsWith('https://'),'mode '+id);}change('研究搜索类型','video');change('研究搜索引擎','youtube');await wait();ok(plugin.store.settings.research.selected.video==='youtube'&&history[index].includes('youtube.com'),'engine');});
await test('Research settings use category choices with persistent checkmarks before showing the editor',()=>{change('研究搜索引擎','__settings__');click('选择研究引擎 bilibili');ok(plugin.store.settings.research.selected.video==='bilibili','choice not persisted');ok(control('选择研究引擎 bilibili').getAttribute('aria-checked')==='true','selected state missing');ok((document.querySelector('.ms-research-editor') as HTMLElement).hidden,'editor shown before requested');click('关闭研究设置');});
await test('Search engine editor rejects invalid URL and saves valid order/name',()=>{change('研究搜索引擎','__settings__');click('编辑搜索引擎与顺序');control('bilibili 搜索网址').value='file:///%s';control('bilibili 搜索网址').dispatchEvent(new Event('input'));const save=[...document.querySelectorAll('button')].find(b=>b.textContent==='Save')!;save.click();ok(!!document.querySelector('.ms-research-settings-error')?.textContent,'no validation');control('bilibili 搜索网址').value='https://example.test/search?q=%s';control('bilibili 搜索网址').dispatchEvent(new Event('input'));click('上移 text');save.click();ok(plugin.store.settings.research.order[0]==='text','order');ok(plugin.store.settings.research.engines.find((e:any)=>e.id==='bilibili').url.includes('example.test'),'engine save');});
await test('New research window uses the current URL',()=>{click('研究网页操作');const item=[...document.querySelectorAll('[role=menuitem]')].find(e=>e.textContent==='在独立研究窗口中打开') as HTMLElement;item.click();ok(opened[0]===handle.currentUrl(),'window URL');});
await test('Web PDF export preserves bytes, registers the document and targets the original study set',async()=>{plugin.store.activeStudySetId='A';handle.navigate('https://example.test/original-article');await wait();pdfOverride=async()=>{handle.navigate('https://example.test/different-article');plugin.store.activeStudySetId='B';return new TextEncoder().encode('%PDF-1.4\nfixed synthetic PDF');};const file=await saveResearchPdf(plugin,port);ok(files.has(file.path)&&recorded.includes(file.path),'PDF not saved');ok(origins.get(file.path)==='https://example.test/original-article','original source lost or changed during printing');ok(sets[0].docPaths.includes(file.path)&&!sets[1].docPaths.includes(file.path),'wrong set');ok(new TextDecoder().decode(files.get(file.path)!.data).includes('fixed synthetic PDF'),'bytes changed');});
await test('Repeated export creates a distinct file and keeps prior bytes intact',async()=>{const previous=[...files.values()][0],before=previous.data.slice(0);const file=await saveResearchPdf(plugin,port,'A');ok(files.size===2&&file.path.endsWith('2.pdf'),'overwritten file');ok(new Uint8Array(before).every((b,i)=>b===new Uint8Array(previous.data)[i]),'prior PDF mutated');});

});
