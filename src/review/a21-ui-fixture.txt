const wait = (ms = 60) => new Promise(done => setTimeout(done, ms));
const assert = (value: any, message: string) => { if (!value) throw new Error(message); };
const same = (actual: any, expected: any, message: string) => assert(JSON.stringify(actual) === JSON.stringify(expected), `${message}: ${JSON.stringify(actual)} != ${JSON.stringify(expected)}`);
const field = <T extends HTMLElement>(label: string): T => {
  const found = document.querySelector<T>(`[aria-label="${label}"]`);
  assert(found, `缺少字段「${label}」`); return found!;
};
function button(name: string): HTMLButtonElement {
  const found = [...document.querySelectorAll<HTMLButtonElement>('button')].find(node => node.isConnected && !node.closest('[hidden]') && (node.getAttribute('aria-label') === name || node.textContent === name));
  assert(found, `缺少按钮「${name}」`); return found!;
}
async function test(name: string, run: () => Promise<void>) {
  try { document.querySelectorAll('.ms-modal-bg').forEach(node => node.remove()); await run(); results.push({ name, passed: true }); }
  catch (error) { results.push({ name, passed: false, error: String(error) }); }
}
async function fresh() {
  let disk: any = {};
  const files = new Map<string, string>();
  const app: any = { vault: { getAbstractFileByPath: () => null, getFiles: () => [], adapter: {
    exists: async (path: string) => files.has(path), mkdir: async (path: string) => files.set(path, ''),
    write: async (path: string, value: string) => files.set(path, value), read: async (path: string) => files.get(path),
  } } };
  const host: any = { app, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } };
  const store = new Store(host); await store.load();
  const set = store.createStudySet('学习集甲', ['a21.pdf']);
  const cards = ['甲', '乙', '丙'].map((name, index) => ({ id: `a21-${index}`, docPath: 'a21.pdf', page: index + 1,
    type: 'flashcard' as const, front: `问题${name}`, back: `答案${name}`, tags: index === 1 ? ['法律/民法'] : ['其他'],
    links: [], favorite: index === 1, createdAt: `2020-01-0${index + 1}T00:00:00.000Z` }));
  for (const card of cards) store.addCard(card);
  const deck = store.createReviewDeck('A21 卡组', set.id);
  store.enqueueReviewInDeck(cards.map(card => card.id), deck.id, store.activeCanvasId, new Date('2020-01-05T00:00:00Z'));
  const plugin: any = { store, app };
  return { store, plugin, host, set, deck, cards };
}
async function open(plugin: any) {
  assert(typeof (review as any).openPractice === 'function', '缺少独立刷题入口');
  await (review as any).openPractice(plugin); await wait();
}
async function submit(answer: string, verdict: string, mistake = '') {
  field<HTMLTextAreaElement>('练习答案').value = answer;
  field<HTMLTextAreaElement>('练习错因').value = mistake;
  button('提交答案').click(); await wait(); button(verdict).click(); await wait();
}
