const results: any[] = [];

const wait = (ms = 60) => new Promise(done => setTimeout(done, ms));
const assert = (value: any, message: string) => { if (!value) throw new Error(message); };
const same = (actual: any, expected: any, message: string) => assert(JSON.stringify(actual) === JSON.stringify(expected), `${message}: ${JSON.stringify(actual)} != ${JSON.stringify(expected)}`);
function button(name: string): HTMLButtonElement {
  const found = [...document.querySelectorAll<HTMLButtonElement>('button')].find(node => node.isConnected && !node.closest('[hidden]') && (node.getAttribute('aria-label') === name || node.textContent === name));
  assert(found, `缺少可用按钮「${name}」`); return found!;
}
async function test(name: string, run: () => Promise<void>) {
  try { document.querySelectorAll('.ms-modal-bg,.ms-review-context-return').forEach(node => node.remove()); await run(); results.push({ name, passed: true }); }
  catch (error) { results.push({ name, passed: false, error: String(error) }); }
}
async function fresh() {
  let disk: any = {};
  const files = new Map<string, string>();
  const source = new TFile('synthetic.pdf');
  const app: any = { vault: { getAbstractFileByPath: (path: string) => path === source.path ? source : null, getFiles: () => [source],
    adapter: { exists: async (path: string) => files.has(path), mkdir: async (path: string) => files.set(path, ''),
      write: async (path: string, value: string) => files.set(path, value), read: async (path: string) => files.get(path) } } };
  const host: any = { app, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } };
  const store = new Store(host); await store.load();
  const set = store.createStudySet('学习集甲', ['synthetic.pdf']);
  const canvasId = store.activeCanvasId!;
  const cards = ['甲', '乙', '丙'].map((name, index) => ({ id: `a20-${index}`, docPath: 'synthetic.pdf', page: index + 1,
    type: 'flashcard' as const, front: `问题${name}`, back: `答案${name}`, tags: index === 1 ? ['父/子'] : ['旧标签'],
    links: [], favorite: index === 1, color: index === 1 ? 'red' : 'blue', createdAt: `2020-01-0${index + 1}T00:00:00.000Z` }));
  for (const card of cards) store.addCard(card);
  const deck = store.createReviewDeck('A20 卡组', set.id);
  store.enqueueReviewInDeck(cards.map(card => card.id), deck.id, canvasId, new Date('2020-01-05T00:00:00Z'));
  const navigations: any[] = [];
  const plugin: any = { store, app, activeReader: null,
    openDocument: async (file: any) => { navigations.push({ kind: 'document', path: file.path }); return { gotoSourcePage: (page: number) => navigations.push({ kind: 'page', page }) }; },
    studyWorkspace: { revealReviewCardContext: async (cardId: string, scopeId: string) => { navigations.push({ kind: 'mindmap', cardId, scopeId }); return true; } } };
  return { store, plugin, host, deck, set, canvasId, cards, navigations };
}

  await test('复习快捷键只在背面评分，输入与无关按键不受影响', async () => {
    for (const grade of [1, 2, 3, 4]) {
      const { store, plugin, deck, cards } = await fresh();
      await startReview(plugin, { deckId: deck.id, presentation: 'score', cardIds: [cards[0].id] }); await wait();
      const key = (value: string, target: HTMLElement = document.body) => {
        const event = new KeyboardEvent('keydown', { key: value, bubbles: true, cancelable: true }); target.dispatchEvent(event); return event;
      };
      const count = () => store.reviewLogs.filter(log => !log.initialization).length;
      for (const value of ['1', '2', '3', '4', 'ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown', 'Enter']) assert(!key(value).defaultPrevented, `${value} 被拦截`);
      assert(count() === 0 && !!document.querySelector('[aria-label="复习问题"]'), '正面按键改变了复习');
      assert(key(' ').defaultPrevented, '空格未翻面'); await wait();
      assert(!!document.querySelector('[aria-label="复习答案"]'), '空格未显示背面');
      key(' '); await wait(); assert(!!document.querySelector('[aria-label="复习问题"]'), '第二次空格未回正面');
      key(' '); await wait();
      button('批注与工具').click();
      const input = document.querySelector<HTMLTextAreaElement>('[aria-label="本次复习批注"]')!;
      input.focus(); const inputEvent = key(String(grade), input);
      assert(!inputEvent.defaultPrevented && count() === 0, '输入框数字被拦截或触发评分');
      document.execCommand('insertText', false, String(grade));
      assert(input.value === String(grade), '输入框不能输入数字');

      assert(count() === 1 && store.getLatestReviewLog()?.rating === grade, `数字 ${grade} 评分错误`);
      assert(document.querySelector('.ms-native-review-completion')?.textContent?.includes('已到最后一张卡片'), '最后一张未进入收尾');
      button('完成').click(); assert(!document.querySelector('.ms-review-score'), '完成未退出复习');
    }
  });

  await test('同一卡在两个卡组按各自上限调度，未设上限保持原算法', async () => {
    const { store, plugin, host, deck, cards, canvasId } = await fresh();
    const other = store.createReviewDeck('A20 长间隔'), unlimited = store.createReviewDeck('A20 无上限');
    const now = new Date('2020-01-05T00:00:00Z');
    store.enqueueReviewInDeck([cards[0].id], other.id, canvasId, now); store.enqueueReviewInDeck([cards[0].id], unlimited.id, canvasId, now);
    openReviewDecks(plugin); await wait();
    const picker = document.querySelector<HTMLSelectElement>('[aria-label="选择复习卡组"]')!; picker.value = deck.id; picker.dispatchEvent(new Event('change', { bubbles: true }));
    const max = document.querySelector<HTMLInputElement>('[aria-label="卡组最长间隔（天）"]')!; max.value = '1'; max.closest('form')!.dispatchEvent(new Event('submit', { bubbles: true, cancelable: true })); await wait();
    assert(store.getReviewDeck(deck.id)?.maxInterval === 1, '卡组设置未保存'); button('关闭').click();
    store.setReviewDeckMaxInterval(other.id, 2);
    let rejected = false; try { store.setReviewDeckMaxInterval(other.id, 0); } catch { rejected = true; }
    assert(rejected && store.getReviewDeck(other.id)?.maxInterval === 2, '无效上限未拒绝');
    const questionA = store.deckQuestionsForCard(cards[0].id, deck.id)[0], questionB = store.deckQuestionsForCard(cards[0].id, other.id)[0], questionC = store.deckQuestionsForCard(cards[0].id, unlimited.id)[0];
    for (const question of [questionA, questionB, questionC]) store.srs[question.id] = { ...store.getReviewState(cards[0].id, question.id),
      due: now.toISOString(), interval: 60, reps: 8, stability: 60, difficulty: 5, state: 2,
      lastReviewed: '2019-11-06T00:00:00.000Z', scheduledDays: 60, elapsedDays: 60, learningSteps: 0 };
    await startReview(plugin, { deckId: deck.id, cardIds: [cards[0].id], presentation: 'score' }); await wait(); button('显示答案').click(); await wait();
    assert(document.querySelector('.ms-grade.easy')?.textContent?.includes('1d'), '评分预览未显示卡组上限'); button('结束复习').click();
    const originalB = structuredClone(store.getReviewState(cards[0].id, questionB.id));
    const first = store.gradeCard(cards[0].id, 4, { questionId: questionA.id, deckId: deck.id, scopeId: canvasId, now });
    same(store.getReviewState(cards[0].id, questionB.id), originalB, '评分甲影响乙的进度');
    const second = store.gradeCard(cards[0].id, 4, { questionId: questionB.id, deckId: other.id, scopeId: canvasId, now });
    const uncapped = store.gradeCard(cards[0].id, 4, { questionId: questionC.id, deckId: unlimited.id, scopeId: canvasId, now });
    assert(first.interval === 1 && second.interval === 2 && uncapped.interval > 2, `上限未独立生效: ${first.interval}, ${second.interval}, ${uncapped.interval}`);
    for (const [state, expected] of [[first, 1], [second, 2]] as const) {
      assert(new Date(state.due).getTime() - now.getTime() === expected * 86400000, '到期时间与上限不一致');
      assert(state.scheduledDays === expected, 'FSRS scheduledDays 与上限不一致');
    }
    same(store.getReviewState(cards[0].id, questionA.id), first, '评分乙影响甲的进度');
    await store.flush(); const reopened = new Store(host); await reopened.load();
    assert(reopened.getReviewDeck(deck.id)?.maxInterval === 1 && reopened.getReviewDeck(other.id)?.maxInterval === 2 && reopened.getReviewDeck(unlimited.id)?.maxInterval === undefined, '重开后上限变化');
    reopened.setReviewDeckMaxInterval(deck.id, undefined); assert(reopened.getReviewDeck(deck.id)?.maxInterval === undefined && reopened.getReviewDeck(other.id)?.maxInterval === 2, '清除上限影响其他卡组');
  });
