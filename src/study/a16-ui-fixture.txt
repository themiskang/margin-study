async function imagePixels(source: string) {
  const image = new Image(); image.src = source; await image.decode();
  const canvas = document.createElement('canvas'); canvas.width = image.naturalWidth; canvas.height = image.naturalHeight;
  const context = canvas.getContext('2d')!; context.drawImage(image, 0, 0);
  return { width: canvas.width, height: canvas.height, pixels: context.getImageData(0, 0, canvas.width, canvas.height).data };
}
function write(root: HTMLElement) {
  const canvas = root.querySelector<HTMLCanvasElement>('canvas')!, rect = canvas.getBoundingClientRect();
  const send = (type: string, x: number, y: number, pressure: number) => canvas.dispatchEvent(new PointerEvent(type, { bubbles: true, cancelable: true, pointerId: 71, pointerType: 'pen', button: 0, buttons: type === 'pointerup' ? 0 : 1, pressure, clientX: rect.left + rect.width * x, clientY: rect.top + rect.height * y }));
  send('pointerdown', .03, .15, .4); send('pointermove', .06, .25, .6); send('pointerup', .08, .35, 0);
}
