(MarkdownRenderer as any).render = async (_app: any, text: string, host: HTMLElement) => {
  host.replaceChildren(); let list: HTMLUListElement | null = null;
  for (const line of text.split('\n')) {
    const heading = /^(#{1,6}) (.*)$/.exec(line), bullet = /^- (.*)$/.exec(line);
    const block = document.createElement(heading ? `h${heading[1].length}` : bullet ? 'li' : 'p');
    for (const part of (heading?.[2] ?? bullet?.[1] ?? line).split(/(\*\*[^*]+\*\*)/g)) {
      if (part.startsWith('**') && part.endsWith('**')) { const strong = document.createElement('strong'); strong.textContent = part.slice(2,-2); block.append(strong); }
      else block.append(document.createTextNode(part));
    }
    if (bullet) { if (!list) { list = document.createElement('ul'); host.append(list); } list.append(block); }
    else { list = null; host.append(block); }
  }
};

async function fresh(existing?: any) {
  let disk: any = {}; const files = new Map<string,string>();
  const pdf = new TFile('Alpha.pdf');
  const app: any = { vault: { getFiles: () => [pdf], getAbstractFileByPath: (path: string) => path === pdf.path ? pdf : null,
    adapter: { exists: async (path: string) => files.has(path), mkdir: async (path: string) => files.set(path,''), write: async (path: string,value: string) => files.set(path,value), read: async (path: string) => files.get(path) } } };
  const host = existing ?? { app, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } };
  const store = new Store(host); await store.load(); stores.push(store);
  const set = store.studySets[0] ?? store.createStudySet('Study'); const canvasId = store.activeCanvasId!;
  let opens = 0; const events: string[] = [], sentinel = { name: 'independent-sidebar' };
  const plugin: any = { store, app: host.app, activeReader: null, sidebar: sentinel, refreshSidebar(){}, refreshMindmap(){}, viewEvents: { onMindmapCardSelect: (id: string) => events.push(id) },
    openDocument: async () => { opens++; return { flashAnnotation(){}, gotoPage(){} }; } };
  const addCard = (id: string, text: string, patch: any = {}) => store.addCard({ id, docPath:'Alpha.pdf', type:'excerpt', front:text, tags:[], links:[], color:'#ffd54f', createdAt:'2026-10-15T01:00:00Z', ...patch });
  const addNode = (id: string, cardId?: string, patch: any = {}) => { store.addNode({id,cardId,canvasId,title:id,parentId:null,order:store.nodes.length,collapsed:false,x:50+store.nodes.length*260,y:90,...patch}); return store.getNode(id)!; };
  const hostElement = () => { const host = document.createElement('div'); host.style.cssText='height:620px;min-width:0'; target.append(host); return host; };
  const outline = (options: any = {}) => { const host=hostElement(), view=new OutlineRenderer(plugin,options); view.mount(host); mounted.push(view); return {host,view}; };
  const board = (options: any = {}) => { const host=hostElement(), view=new CardBoardRenderer(plugin,options); view.mount(host); mounted.push(view); return {host,view}; };
  const graph = (options: any = {}) => { const host=hostElement(), view=new MindmapRenderer(plugin,options); view.mount(host); mounted.push(view); return {host,view}; };
  return {store,plugin,set,canvasId,host,addCard,addNode,outline,board,graph,events,sentinel,opens:()=>opens};
}
const edgeSignature = (edges: any[]) => JSON.stringify(edges.map(edge => ({ from:edge.from, to:edge.to, label:edge.label })));
async function loadWithEdges(p: any, edges: {from:string;to:string;label?:string}[]) {
  p.store.edges = structuredClone(edges);
  await p.store.flush();
  const loaded = await fresh(p.host);
  assert(edgeSignature(loaded.store.edges) === edgeSignature(edges), `synthetic legacy edge fixture was not loaded: ${edgeSignature(loaded.store.edges)}`);
  return loaded;
}
async function parentChild(edge?: {from:string;to:string;label?:string}, childFirst = false) {
  const p = await fresh(); p.addCard('parent','Parent'); p.addCard('child','Child');
  p.addNode('parent-node','parent',{order:childFirst?1:0}); p.addNode('child-node','child',{parentId:'parent-node',order:childFirst?0:1});
  return edge ? loadWithEdges(p,[edge]) : loadWithEdges(p,[]);
}
