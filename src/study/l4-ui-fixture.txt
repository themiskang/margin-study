const encode = (text: string) => text.replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;');
(MarkdownRenderer as any).render = async (_app: any, text: string, host: HTMLElement) => {
  host.innerHTML = encode(text).replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');

function button(root: ParentNode, name: string): HTMLButtonElement {
  const found = [...root.querySelectorAll<HTMLButtonElement>('button')].find(node => node.getAttribute('aria-label') === name || node.textContent === name);
  assert(found, `Missing button ${name}`); return found!;
}
const action = (root: ParentNode, name: string) => root.querySelector<HTMLButtonElement>(`.ms-selection-actions [data-action="${name}"]`)!;
const card = (root: ParentNode, id: string) => root.querySelector<HTMLElement>(`.ms-node[data-id="${id}"]`)!;
async function test(name: string, run: () => any) {
  try {
    document.querySelectorAll('.ms-modal-bg,#test-menu').forEach(node => node.remove());
    target.replaceChildren(); target.style.width = ''; await run(); results.push({ name, passed: true });
  } catch (error) { results.push({ name, passed: false, error: String(error) }); }
}
async function fresh(host?: any) {
  let disk: any = {};
  const files = new Map<string, string>();
  const source = new TFile('synthetic.pdf');
  const app: any = { vault: { getAbstractFileByPath: (path: string) => path === source.path ? source : null, getFiles: () => [source],
    adapter: { exists: async (path: string) => files.has(path), mkdir: async (path: string) => files.set(path, ''),
      write: async (path: string, value: string) => files.set(path, value), read: async (path: string) => files.get(path) } } };
  host ??= { app, loadData: async () => structuredClone(disk), saveData: async (value: any) => { disk = structuredClone(value); } };
  const store = new Store(host); await store.load(); stores.push(store);
  const set = store.studySets[0] ?? store.createStudySet('L4 synthetic set');
  const canvasId = store.activeCanvasId!;
  let opens = 0; const jumps: any[] = [];
  const plugin: any = { store, app: host.app, activeReader: null, refreshSidebar() {}, refreshMindmap() {}, sidebar: null,
    openDocument: async () => { opens++; return { flashAnnotation: (id: string) => jumps.push(['anchor', id]), gotoPage: (page: number) => jumps.push(['page', page]) }; } };
  const addCard = (id: string, text: string, patch: any = {}) => store.addCard({ id, docPath: 'synthetic.pdf', page: 1,
    type: 'excerpt', front: text, color: '#ffffa0', tags: [], links: [], createdAt: '2026-10-15T00:00:00Z', ...patch });
  const addNode = (id: string, cardId?: string, patch: any = {}) => {
    store.addNode({ id, cardId, canvasId, parentId: null, title: `local-${id}`, order: store.nodes.length,
      collapsed: false, x: 90 + store.nodes.length * 290, y: 100, ...patch }); return store.getNode(id)!;
  };
  const mount = () => { const wrap = document.createElement('div'); wrap.style.height = '650px'; target.append(wrap);
    const renderer = new MindmapRenderer(plugin); renderer.mount(wrap); renderers.push(renderer); return { renderer, wrap }; };
  return { store, plugin, set, canvasId, addCard, addNode, mount, host, jumps, opens: () => opens };
}
