import { openGlobalSearch } from '__PROJECT__/src/search/globalSearch';

const results: any[] = [];
(window as any).results = results;
const assert = (value: unknown, message: string) => { if (!value) throw new Error(message); };
const test = async (name: string, action: () => Promise<void>) => { try { await action(); results.push({name,passed:true}); } catch (error) { results.push({name,passed:false,error:String(error)}); } };
const wait = () => new Promise(resolve => setTimeout(resolve, 350));
const body = document.createElement('div'); body.textContent = '正文 contains target content';
let navigation = '', selectedCanvas = '', command = false;
const reader: any = { getFilePath:()=> 'example.pdf', getScrollContainer:()=>body,
  getEditableOutline:async()=>[{title:'目标章节',page:2,level:0}], navigateToOutline:(item:any)=>navigation=item.title,
  getDoc:()=>({numPages:2,getPage:async(page:number)=>({getTextContent:async()=>({items:[{str:page===1?'unrelated page':'target document body'}]})})}),
  gotoPage:(page:number)=>navigation=String(page) };
const plugin: any = { activeReader:reader, manifest:{id:'margin-study'}, refreshMindmap(){},
  app:{commands:{commands:{'margin-study:open-cards':{callback:()=>command=true}}},vault:{getFiles:()=>[],getAbstractFileByPath:()=>null}},
  store:{nodes:[{id:'node',canvasId:'canvas',title:'target mindmap'}],cards:[],canvases:[{id:'canvas',name:'Map',studySetId:'set'}],documents:[],activeStudySetId:'set',getCard:()=>null,setActiveCanvas:(id:string)=>selectedCanvas=id,setActiveStudySet(){},loadAnnotations:async()=>[]} };
const tab = (name:string) => [...document.querySelectorAll<HTMLButtonElement>('[role=tab]')].find(item=>item.textContent===name)!;
const query = async (value:string) => { const input=document.querySelector<HTMLInputElement>('[aria-label="搜索关键词"]')!; input.value=value; input.dispatchEvent(new Event('input')); await wait(); };
const hit = () => document.querySelector<HTMLButtonElement>('.ms-search-result')!;

try {
  await test('four native categories are selectable and cards/annotations remain available', async()=>{
    openGlobalSearch(plugin);
    assert([...document.querySelectorAll('[role=tab]')].map(item=>item.textContent).join('|')==='脑图|文档目录|文档正文|功能|卡片与批注','categories differ');
    await query('target'); assert(hit()?.textContent?.includes('mindmap'),'mindmap result missing'); hit().click(); assert(selectedCanvas==='canvas','mindmap result did not switch canvas');
  });
  await test('directory and document-body searches use real reader content and navigate', async()=>{
    openGlobalSearch(plugin); tab('文档目录').click(); await query('目标'); assert(hit()?.textContent?.includes('目标章节'),'outline result missing'); hit().click(); assert(navigation==='目标章节','outline did not navigate');
    openGlobalSearch(plugin); tab('文档正文').click(); await query('target'); assert(hit()?.textContent?.includes('第 2 页'),'document content result missing'); hit().click(); assert(navigation==='2','body result did not navigate to page');
  });
  await test('feature category executes matching indexed command', async()=>{
    openGlobalSearch(plugin); tab('功能').click(); await query('卡片管理'); assert(hit()?.textContent?.includes('卡片管理'),'feature result missing'); hit().click(); assert(command,'feature command not executed');
  });
} finally { document.querySelectorAll('.ms-modal-bg').forEach(item=>item.remove()); (window as any).done=true; }
